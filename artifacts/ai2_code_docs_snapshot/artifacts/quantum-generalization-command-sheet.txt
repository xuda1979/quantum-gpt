# Strict unseen quantum generalization command sheet
# dataset train: data/generated/omnicoder-quantum-generalization-holdout-v1/train.jsonl
# dataset eval: data/generated/omnicoder-quantum-generalization-holdout-v1/eval.jsonl
# benchmark: evals/benchmarks/quantum_generalization_holdout_v1.txt

## SFT
python3 training/qwen_sft_peft.py \
  --model-name models/OmniCoder-9B \
  --train-file data/generated/omnicoder-quantum-generalization-holdout-v1/train.jsonl \
  --eval-file data/generated/omnicoder-quantum-generalization-holdout-v1/eval.jsonl \
  --output-dir outputs/omnicoder9b-quantum-generalization-sft-v1 \
  --device npu \
  --max-steps 40 \
  --num-epochs 2 \
  --per-device-batch-size 1 \
  --gradient-accumulation-steps 8 \
  --learning-rate 2e-4 \
  --train-on-completions-only

## Unseen Eval
python3 scripts/run_hf_pass1_eval.py \
  --run-dir evals/runs/omnicoder-quantum-generalization-holdout-v1-clean \
  --base-model models/OmniCoder-9B \
  --adapter outputs/omnicoder9b-quantum-generalization-sft-v1/adapter \
  --device npu \
  --score

## GRPO
python3 training/grpo_trainer.py \
  --model-name models/OmniCoder-9B \
  --adapter-init outputs/omnicoder9b-quantum-generalization-sft-v1/adapter \
  --benchmark-file evals/benchmarks/quantum_generalization_holdout_v1.txt \
  --domain-filter quantum \
  --output-dir outputs/omnicoder9b-quantum-generalization-grpo-v2 \
  --device npu \
  --group-size 4 \
  --grpo-steps 16 \
  --max-new-tokens 512 \
  --max-seq-length 4096 \
  --reward-pass-weight 0.6 \
  --reward-syntax-weight 0.1 \
  --reward-interface-weight 0.15 \
  --reward-verifier-weight 0.15 \
  --quantum-priority 1.5 \
  --curriculum-uncertainty-bonus 0.35 \
  --temperature 0.8
