import importlib.util
import math


def _load(candidate_path: str):
    spec = importlib.util.spec_from_file_location("candidate", candidate_path)
    module = importlib.util.module_from_spec(spec)
    assert spec.loader is not None
    spec.loader.exec_module(module)
    return module


def _close(a, b, tol=1e-9):
    return abs(a - b) < tol


def _mat_close(a, b, tol=1e-6):
    if len(a) != len(b):
        return False
    for ra, rb in zip(a, b):
        if len(ra) != len(rb):
            return False
        for x, y in zip(ra, rb):
            if abs(x - y) > tol:
                return False
    return True


def run_tests(candidate_path: str) -> dict:
    module = _load(candidate_path)
    failures = []

    # --- depolarizing_channel ---
    # rho' = (1-p)*rho + (p/3)*(X*rho*X + Y*rho*Y + Z*rho*Z)
    # For p=0, output = input
    rho_0 = [[1.0, 0.0], [0.0, 0.0]]
    out_0 = module.depolarizing_channel(rho_0, 0.0)
    if not _mat_close(out_0, rho_0):
        failures.append(f"depolarizing(|0><0|, p=0) != identity: {out_0}")

    # For p=1 (fully depolarizing), output = I/2 regardless of input
    out_full = module.depolarizing_channel(rho_0, 1.0)
    expected_full = [[0.5, 0.0], [0.0, 0.5]]
    if not _mat_close(out_full, expected_full, tol=1e-6):
        failures.append(f"depolarizing(|0><0|, p=1) = {out_full}, expected I/2")

    # For p=0.5, rho_0 -> 0.5*|0><0| + 0.5*(I/2) = [[0.75, 0], [0, 0.25]]
    out_half = module.depolarizing_channel(rho_0, 0.5)
    expected_half = [[0.75, 0.0], [0.0, 0.25]]
    if not _mat_close(out_half, expected_half, tol=1e-6):
        failures.append(f"depolarizing(|0><0|, p=0.5) = {out_half}, expected {expected_half}")

    # Check with |+><+| state
    rho_plus = [[0.5, 0.5], [0.5, 0.5]]
    out_plus = module.depolarizing_channel(rho_plus, 0.5)
    expected_plus = [[0.5, 0.25], [0.25, 0.5]]
    if not _mat_close(out_plus, expected_plus, tol=1e-6):
        failures.append(f"depolarizing(|+><+|, p=0.5) = {out_plus}, expected {expected_plus}")

    # --- amplitude_damping_channel ---
    # Kraus: K0 = [[1, 0], [0, sqrt(1-gamma)]], K1 = [[0, sqrt(gamma)], [0, 0]]
    # rho' = K0 rho K0^dag + K1 rho K1^dag

    # gamma=0 -> identity
    out_ad0 = module.amplitude_damping_channel(rho_plus, 0.0)
    if not _mat_close(out_ad0, rho_plus):
        failures.append(f"amplitude_damping(|+><+|, gamma=0) != input: {out_ad0}")

    # gamma=1 -> everything goes to |0>
    rho_1 = [[0.0, 0.0], [0.0, 1.0]]
    out_ad1 = module.amplitude_damping_channel(rho_1, 1.0)
    expected_ad1 = [[1.0, 0.0], [0.0, 0.0]]
    if not _mat_close(out_ad1, expected_ad1, tol=1e-6):
        failures.append(f"amplitude_damping(|1><1|, gamma=1) = {out_ad1}, expected |0><0|")

    # gamma=0.5 on |1><1|: K0 rho K0^dag = [[0,0],[0,0.5]], K1 rho K1^dag = [[0.5,0],[0,0]]
    out_ad_half = module.amplitude_damping_channel(rho_1, 0.5)
    expected_ad_half = [[0.5, 0.0], [0.0, 0.5]]
    if not _mat_close(out_ad_half, expected_ad_half, tol=1e-6):
        failures.append(f"amplitude_damping(|1><1|, gamma=0.5) = {out_ad_half}, expected {expected_ad_half}")

    # --- channel_fidelity ---
    # F(rho, sigma) = (Tr(sqrt(sqrt(rho) sigma sqrt(rho))))^2
    # For pure states: F(|0><0|, |0><0|) = 1
    f_same = module.channel_fidelity(rho_0, rho_0)
    if not _close(f_same, 1.0, tol=1e-6):
        failures.append(f"fidelity(|0><0|, |0><0|) = {f_same}, expected 1.0")

    # F(|0><0|, |1><1|) = 0
    f_orth = module.channel_fidelity(rho_0, rho_1)
    if not _close(f_orth, 0.0, tol=1e-6):
        failures.append(f"fidelity(|0><0|, |1><1|) = {f_orth}, expected 0.0")

    # F(|0><0|, I/2) = 0.5
    rho_mixed = [[0.5, 0.0], [0.0, 0.5]]
    f_mixed = module.channel_fidelity(rho_0, rho_mixed)
    if not _close(f_mixed, 0.5, tol=1e-6):
        failures.append(f"fidelity(|0><0|, I/2) = {f_mixed}, expected 0.5")

    # Trace preservation check
    out_check = module.depolarizing_channel(rho_plus, 0.3)
    tr = out_check[0][0] + out_check[1][1]
    if not _close(tr, 1.0, tol=1e-6):
        failures.append(f"depolarizing trace = {tr}, expected 1.0")

    return {
        "passed": not failures,
        "details": failures or ["Quantum channel simulations correct for all test cases"],
    }
