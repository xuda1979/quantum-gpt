# Variational Quantum Eigensolver (VQE)

## Concept
VQE estimates the lowest eigenvalue of a Hermitian operator H using a
parameterised circuit ansatz |psi(theta)> and a classical optimiser:

  E(theta) = <psi(theta)| H |psi(theta)>

The optimiser minimises E(theta) over theta. The result is an upper
bound on the true ground-state energy by the variational principle.

1. Encode H as a sum of Pauli strings: H = sum_i c_i P_i.

import importlib.util
import math


def _load(candidate_path: str):
    spec = importlib.util.spec_from_file_location("candidate", candidate_path)
    module = importlib.util.module_from_spec(spec)
    assert spec.loader is not None
    spec.loader.exec_module(module)
    return module


def run_tests(candidate_path: str) -> dict:
    module = _load(candidate_path)
    failures = []

    # ry_state tests
    a, b = module.ry_state(0.0)
    if not (math.isclose(a, 1.0) and math.isclose(b, 0.0)):
        failures.append(f"ry_state(0) -> ({a}, {b}), expected (1, 0)")

    a, b = module.ry_state(math.pi)
    if not (math.isclose(a, 0.0, abs_tol=1e-9) and math.isclose(b, 1.0)):
        failures.append(f"ry_state(pi) -> ({a}, {b}), expected (0, 1)")

    # energy_expectation: H = Z (eigenvalues +1, -1)
    e0 = module.energy_expectation(0.0, (0.0, 1.0, 0.0))  # |0> -> <Z>=+1
    if not math.isclose(e0, 1.0, abs_tol=1e-9):
        failures.append(f"energy(0, Z) -> {e0}, expected 1.0")

    e1 = module.energy_expectation(math.pi, (0.0, 1.0, 0.0))  # |1> -> <Z>=-1
    if not math.isclose(e1, -1.0, abs_tol=1e-9):
        failures.append(f"energy(pi, Z) -> {e1}, expected -1.0")

    # energy_expectation: H = X (eigenvalues +1, -1)
    e_x = module.energy_expectation(math.pi / 2, (0.0, 0.0, 1.0))  # RY(pi/2)|0> = |+>
    if not math.isclose(e_x, 1.0, abs_tol=1e-9):
        failures.append(f"energy(pi/2, X) -> {e_x}, expected 1.0")

    # H = I + 0.5*Z + 0.3*X at theta=0: E = 1 + 0.5*1 + 0.3*0 = 1.5
    e_combo = module.energy_expectation(0.0, (1.0, 0.5, 0.3))
    if not math.isclose(e_combo, 1.5, abs_tol=1e-9):
        failures.append(f"energy(0, (1,0.5,0.3)) -> {e_combo}, expected 1.5")

    # find_min_energy: H = Z should find theta ~ pi (E = -1)
    best_theta, best_e = module.find_min_energy((0.0, 1.0, 0.0))
    if not math.isclose(best_e, -1.0, abs_tol=0.05):
        failures.append(f"find_min_energy(Z) -> E={best_e}, expected ~-1.0")

    # find_min_energy: H = 0.5*Z + 0.5*X, ground state energy = -1/sqrt(2) ~ -0.707
    _, best_e2 = module.find_min_energy((0.0, 0.5, 0.5))
    expected_gs = -math.sqrt(0.5)
    if not math.isclose(best_e2, expected_gs, abs_tol=0.05):
        failures.append(f"find_min_energy(0.5Z+0.5X) -> E={best_e2}, expected ~{expected_gs:.4f}")

    return {
        "passed": not failures,
        "details": failures or ["VQE energy minimization correct for all test cases"],
    }
